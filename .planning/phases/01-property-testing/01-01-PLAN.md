```rust
// Handle ABI — encode/decode are the primary roundtrip target
pub struct Handle {
    pub index: u32,
    pub slot_tag: SlotTag,
}
pub const MAX_HANDLE_INDEX: u32 = 0xFFFF;

impl Handle {
    pub const fn encode(self) -> u64 {
        (self.index as u64 & 0xFFFF) | ((self.slot_tag.0 & 0xFFFF_FFFF_FFFF) << 16)
    }
    pub const fn decode(raw: u64) -> Handle {
        Handle {
            index: (raw & 0xFFFF) as u32,
            slot_tag: SlotTag((raw >> 16) & 0xFFFF_FFFF_FFFF),
        }
    }
}

// SlotTag — 48-bit ABI matching
pub struct SlotTag(pub u64);
impl SlotTag {
    pub const fn abi_matches(self, other: SlotTag) -> bool {
        (self.0 & 0xFFFF_FFFF_FFFF) == (other.0 & 0xFFFF_FFFF_FFFF)
    }
}

// Rights — bitmask with 14 bits across u16
pub struct Rights(u16);
impl Rights {
    pub const fn empty() -> Rights { Rights(0) }
    pub const fn contains(self, other: Rights) -> bool { (self.0 & other.0) == other.0 }
    pub const fn union(self, other: Rights) -> Rights { Rights(self.0 | other.0) }
    pub const fn attenuate(self, mask: Rights) -> Rights { Rights(self.0 & mask.0) }
    pub const fn bits(self) -> u16 { self.0 }
    pub const fn from_bits(bits: u16) -> Rights { Rights(bits) }
}

// ObjectType — five variants, exhaustive
pub enum ObjectType { Space, Time, Field, Observer, Pulsar }

// Badge — u64 wrapper
pub struct Badge(pub u64);

// Per-type complete masks
// SPACE_ALL: 4 bits, TIME_ALL: 2 bits, FIELD_ALL: 6 bits, OBSERVER_ALL: 9 bits, PULSAR_ALL: 2 bits
// Total unique bits across all masks: 14

// Sentinel
pub const CAP_ABSENT: u64 = u64::MAX;
```

    PROP-01 (Handle encode/decode roundtrip):
    - For any index in 0..=MAX_HANDLE_INDEX and any slot_tag in 0..=0xFFFF_FFFF_FFFF: Handle { index, slot_tag: SlotTag(slot_tag) }.encode() decoded back yields the same index and slot_tag
    - For any u64 value: Handle::decode(v).encode() == v when v is a valid encoded handle (index <= MAX_HANDLE_INDEX, tag in 48-bit range)
    - SlotTag::abi_matches is reflexive: any SlotTag matches itself
    - SlotTag::abi_matches ignores bits above 48: SlotTag(x).abi_matches(SlotTag(x | high_bits)) is true when high_bits only set bits 48..63

    PROP-03 (Rights bitmask algebraic laws):
    - Idempotent union: a.union(a) == a
    - Commutative union: a.union(b) == b.union(a)
    - Associative union: a.union(b).union(c) == a.union(b.union(c))
    - Identity: a.union(Rights::empty()) == a
    - Attenuate is intersection: a.attenuate(b) == Rights::from_bits(a.bits() & b.bits())
    - Attenuate can only remove: a.attenuate(b).bits() <= a.bits() (subset property)
    - Contains after union: a.union(b).contains(a) is always true
    - Contains is reflexive: a.contains(a) is always true
    - Empty contains nothing except empty: Rights::empty().contains(a) implies a == Rights::empty()
    - Per-type mask completeness: each *_ALL mask contains every individual right for that type

    PROP-05 (Slot arithmetic):
    - For any index in 0..=MAX_HANDLE_INDEX: Handle { index, slot_tag: SlotTag(0) }.encode() & 0xFFFF == index as u64
    - Encoded handle index bits never exceed 16 bits: for any valid Handle, encode() & 0xFFFF <= MAX_HANDLE_INDEX as u64
    - For any index > MAX_HANDLE_INDEX: encode truncates to lower 16 bits (information loss — this is the boundary condition to document, not prevent)
    - CAP_ABSENT decoded does not produce a valid handle index: Handle::decode(CAP_ABSENT).index == 0xFFFF (MAX_HANDLE_INDEX) — exercise to confirm sentinel behavior
